cmake_minimum_required(VERSION 3.31)
project(MeshDependencyFixture LANGUAGES CXX)
include("${FCG_SOURCE}/cmake/CPM.cmake")
include("${DEPENDENCIES}")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
if(MODE STREQUAL "source")
    set(CPM_DOWNLOAD_ALL ON)
    set(CPM_OpenMesh_SOURCE "${OPENMESH_SOURCE}")
    set(CPM_tinyobjloader_SOURCE "${TINYOBJ_SOURCE}")
    set(CPM_tinyply_SOURCE "${TINYPLY_SOURCE}")
elseif(MODE STREQUAL "local-missing")
    set(CPM_LOCAL_PACKAGES_ONLY ON)
    set(CMAKE_DISABLE_FIND_PACKAGE_OpenMesh ON)
else()
    # Unversioned OpenMesh config, with decoder version metadata deliberately older than the needed API.
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/package")
    set(names OpenMeshCore tinyobjloader::tinyobjloader tinyply)
    set(packages OpenMesh tinyobjloader tinyply)
    list(GET mesh_includes 0 openmesh_headers)
    list(GET mesh_tinyobj_includes 0 tinyobj_headers)
    list(GET mesh_tinyply_includes 0 tinyply_headers)
    set(headers "${openmesh_headers}" "${tinyobj_headers}" "${tinyply_headers}")
    foreach(i RANGE 0 2)
        list(GET names ${i} name)
        list(GET packages ${i} package)
        list(GET headers ${i} includes)
        list(GET mesh_libraries ${i} binary)
        set(config "add_library(${name} UNKNOWN IMPORTED)\nset_target_properties(${name} PROPERTIES IMPORTED_LOCATION \"${binary}\" INTERFACE_INCLUDE_DIRECTORIES \"${includes}\")\nset(${package}_FOUND TRUE)\n")
        if(MODE STREQUAL "provided")
            file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/provided-${i}.cmake" "${config}")
            include("${CMAKE_CURRENT_BINARY_DIR}/provided-${i}.cmake")
        else()
            file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/package/${package}Config.cmake" "${config}")
            set(${package}_DIR "${CMAKE_CURRENT_BINARY_DIR}/package")
        endif()
    endforeach()
    set(CPM_LOCAL_PACKAGES_ONLY ON)
endif()
include("${FCG_SOURCE}/cmake/Dependencies/OpenMesh.cmake")
include("${FCG_SOURCE}/cmake/Dependencies/MeshDecoders.cmake")
foreach(adapter fcg-openmesh fcg-tinyobjloader fcg-tinyply)
    if(NOT TARGET ${adapter})
        message(FATAL_ERROR "Missing ${adapter}")
    endif()
endforeach()
if(MODE STREQUAL "source")
    if(NOT FCG_OPENMESH_BUILT_HERE OR NOT FCG_tinyobjloader_BUILT_HERE OR NOT FCG_tinyply_BUILT_HERE)
        message(FATAL_ERROR "Source provenance missing")
    endif()
    if(DEFINED BUILD_APPS OR DEFINED OPENMESH_DOCS OR BUILD_SHARED_LIBS OR NOT CMAKE_CXX_VISIBILITY_PRESET STREQUAL "hidden")
        message(FATAL_ERROR "Dependency build settings leaked")
    endif()
elseif(FCG_OPENMESH_BUILT_HERE OR FCG_tinyobjloader_BUILT_HERE OR FCG_tinyply_BUILT_HERE)
    message(FATAL_ERROR "Installed dependencies were unnecessarily fetched")
endif()
