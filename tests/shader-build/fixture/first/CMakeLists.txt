# These failures must be diagnosed during configuration, before the shader compiler is invoked.
if (FAILURE_CASE STREQUAL "name")
	fcg_add_shader(../escape SOURCES work.comp.glsl)
elseif (FAILURE_CASE STREQUAL "missing")
	fcg_add_shader(missing SOURCES missing.comp.glsl)
elseif (FAILURE_CASE STREQUAL "suffix")
	fcg_add_shader(bad SOURCES common.glsl)
elseif (FAILURE_CASE STREQUAL "stage")
	fcg_add_shader(duplicate SOURCES work.comp.glsl work.comp.glsl)
elseif (FAILURE_CASE STREQUAL "sources")
	fcg_add_shader(empty)
endif ()

fcg_add_shader(logical SOURCES work.comp.glsl)
if (FAILURE_CASE STREQUAL "duplicate")
	fcg_add_shader(logical SOURCES work.comp.glsl)
endif ()
if (INCLUDE_EXTRA)
	fcg_add_shader(obsolete SOURCES work.comp.glsl)
endif ()
fcg_finalize_shaders(TARGET first-resources NAMESPACE first::embedded)
