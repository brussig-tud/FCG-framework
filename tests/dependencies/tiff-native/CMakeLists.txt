
cmake_minimum_required(VERSION 3.31)
project(TiffNativeFixture LANGUAGES NONE)

# Simulate native-backend selection without changing compiler/platform behavior in unrelated dependencies.
macro(CPMFindPackage)
    message(FATAL_ERROR "Native TIFF decoding attempted package resolution")
endmacro()
set(FCG_LIBTIFF_BUILT_HERE OFF)
if(MODE STREQUAL "imageio-default")
    set(APPLE TRUE)
    unset(SDLIMAGE_BACKEND_IMAGEIO)
elseif(MODE STREQUAL "imageio")
    set(APPLE TRUE)
    set(SDLIMAGE_BACKEND_IMAGEIO ON)
elseif(MODE STREQUAL "wic")
    set(WIN32 TRUE)
    set(SDLIMAGE_BACKEND_WIC ON)
else()
    message(FATAL_ERROR "Unknown native TIFF fixture mode: ${MODE}")
endif()
include("${FCG_SOURCE}/cmake/Dependencies/libtiff.cmake")
if(FCG_LIBTIFF_BUILT_HERE OR TARGET tiff)
    message(FATAL_ERROR "Native TIFF decoding resolved libtiff")
endif()
