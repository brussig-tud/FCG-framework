cmake_minimum_required(VERSION 3.31)
project(DependencyFixture LANGUAGES C CXX)
include(GenerateExportHeader)
set(FCG_PROJECT_NAME FCG-framework)
set(FCG_FRAMEWORK_VERSION 0.0.1)
set(FCG_FRAMEWORK_SOVERSION 0)
set(FCG_CXX_STANDARD cxx_std_23)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_DEBUG_POSTFIX d)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
set(FCG_USE_SHARED_SDL "${TEST_BACKEND_SHARED}")
set(BUILD_SHARED_LIBS "${FRAMEWORK_SHARED}")
set(CPM_USE_LOCAL_PACKAGES OFF)
set(CPM_SDL3_image_SOURCE "${IMAGE_DEPENDENCY_SOURCE}")
set(CPM_libavif_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/must-not-fetch-libavif")
find_package(SDL3 3.4 CONFIG REQUIRED)

# Provide resolved runtime dependencies so resolution must reuse them without lookup or fetching.
add_library(provided_glm INTERFACE)
add_library(glm::glm ALIAS provided_glm)
target_include_directories(provided_glm INTERFACE "${GLM_SOURCE}")
if(MODE STREQUAL "provided-namespaced")
    add_library(provided_imgui UNKNOWN IMPORTED)
    set(imgui_fixture_target provided_imgui)
    add_library(imgui::imgui ALIAS provided_imgui)
else()
    # Also cover normalization of an unnamespaced target into imgui::imgui.
    add_library(imgui UNKNOWN IMPORTED)
    set(imgui_fixture_target imgui)
endif()
set_target_properties(${imgui_fixture_target} PROPERTIES
    IMPORTED_LOCATION "${IMGUI_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${IMGUI_SOURCE};${IMGUI_SOURCE}/backends")
if(MODE MATCHES "^provided")
    if(MODE STREQUAL "provided-imported")
        add_library(provided_image INTERFACE IMPORTED)
    else()
        add_library(provided_image INTERFACE)
    endif()
    add_library(SDL3_image::SDL3_image ALIAS provided_image)
    set(SDLIMAGE_WEBP ON)
    set(SDLIMAGE_AVIF ON)
    if(MODE STREQUAL "provided-empty")
        set(FCG_SDL_IMAGE_FORMATS "")
    elseif(MODE STREQUAL "provided-invalid")
        set(FCG_SDL_IMAGE_FORMATS PNG BANANA)
    elseif(NOT MODE STREQUAL "provided-missing")
        set(FCG_SDL_IMAGE_FORMATS BMP PNG ANI)
    endif()
elseif(MODE MATCHES "^package")
    # A real config-package lookup through CPM loses non-cache variables inside its function scopes.
    # This fixture deliberately reports values different from the caller's requested options.
    set(package_dir "${CMAKE_CURRENT_BINARY_DIR}/package")
    file(MAKE_DIRECTORY "${package_dir}")
    set(package_config "set(SDL3_image_FOUND TRUE)\nif(NOT TARGET SDL3_image::SDL3_image)\nadd_library(SDL3_image::SDL3_image INTERFACE IMPORTED)\nendif()\n")
    foreach(codec AVIF BMP GIF JPG JXL LBM PCX PNG PNM QOI SVG TGA TIF WEBP XCF XPM XV)
        string(APPEND package_config "set(SDLIMAGE_${codec} OFF)\n")
    endforeach()
    string(APPEND package_config "set(SDLIMAGE_PNG ON)\nset(SDLIMAGE_WEBP ON)\n")
    file(WRITE "${package_dir}/SDL3_imageConfig.cmake" "${package_config}")
    file(WRITE "${package_dir}/SDL3_imageConfigVersion.cmake" "set(PACKAGE_VERSION 3.4.4)\nset(PACKAGE_VERSION_COMPATIBLE TRUE)\n")
    set(SDL3_image_DIR "${package_dir}" CACHE PATH "Fixture config package" FORCE)
    set(SDLIMAGE_WEBP OFF)
    set(SDLIMAGE_AVIF ON)
    set(SDLIMAGE_BMP ON)
    if(MODE STREQUAL "package-visible")
        find_package(SDL3_image CONFIG REQUIRED)
    endif()
elseif(MODE MATCHES "^avif-")
    include("${CMAKE_CURRENT_LIST_DIR}/avif.cmake")
elseif(MODE STREQUAL "normal")
    set(SDLIMAGE_WEBP ON)
    set(SDLIMAGE_STRICT ON)
elseif(MODE STREQUAL "cache")
    set(SDLIMAGE_WEBP ON CACHE BOOL "Exercise a parent's cached codec option")
    set(SDLIMAGE_STRICT ON CACHE BOOL "Require the opted-in codec")
elseif(MODE STREQUAL "png-normal")
    set(SDLIMAGE_PNG OFF)
elseif(MODE STREQUAL "png-cache")
    set(SDLIMAGE_PNG OFF CACHE BOOL "Disable a built-in codec without external prerequisites")
elseif(MODE STREQUAL "disabled")
    set(SDLIMAGE_PNG OFF)
    set(SDLIMAGE_TGA OFF)
    set(SDLIMAGE_BMP OFF)
elseif(MODE STREQUAL "unresolved" OR MODE STREQUAL "unresolved-strict")
    set(SDLIMAGE_AVIF ON)
    set(SDLIMAGE_AVIF_SHARED OFF)
    if(MODE STREQUAL "unresolved-strict")
        set(SDLIMAGE_STRICT ON)
    else()
        set(SDLIMAGE_STRICT OFF)
    endif()
    set(CMAKE_DISABLE_FIND_PACKAGE_libavif TRUE)
elseif(NOT MODE STREQUAL "minimal")
    message(FATAL_ERROR "Unknown fixture mode: ${MODE}")
endif()

# Inherited values must not contaminate provenance reported by this configure's resolution step.
set(FCG_SDL_IMAGE_BUILT_HERE stale)
set(FCG_IMGUI_BUILT_HERE stale)
include("${FCG_SOURCE}/cmake/Dependencies.cmake")
if(TARGET Image OR TARGET Core)
    message(FATAL_ERROR "Dependency startup configured a framework library")
endif()
if(NOT FCG_IMGUI_BUILT_HERE STREQUAL "OFF")
    message(FATAL_ERROR "Provided ImGui was marked as built here")
endif()
if(MODE MATCHES "^provided")
    if(NOT FCG_SDL_IMAGE_BUILT_HERE STREQUAL "OFF" OR NOT SDLIMAGE_WEBP)
        message(FATAL_ERROR "Provided SDL_image was reconfigured or marked as built here")
    endif()
elseif(MODE MATCHES "^package")
    if(NOT FCG_SDL_IMAGE_BUILT_HERE STREQUAL "OFF")
        message(FATAL_ERROR "Imported SDL_image was marked as built here")
    endif()
elseif(NOT FCG_SDL_IMAGE_BUILT_HERE STREQUAL "ON")
    message(FATAL_ERROR "Source-built SDL_image provenance was lost")
endif()
if(NOT "${BUILD_SHARED_LIBS}" STREQUAL "${FRAMEWORK_SHARED}" OR NOT CMAKE_DEBUG_POSTFIX STREQUAL "d")
    message(FATAL_ERROR "Dependency options leaked into framework build settings")
endif()
if(MODE MATCHES "^provided|^package" OR MODE STREQUAL "minimal" OR MODE STREQUAL "unresolved")
    if(TARGET avif OR FCG_LIBAVIF_BUILT_HERE)
        message(FATAL_ERROR "Unneeded libavif dependency was resolved")
    endif()
endif()
if(MODE MATCHES "^avif-")
    if(NOT MODE MATCHES "^avif-real|^avif-package" AND NOT MODE STREQUAL "avif-provided" AND NOT MODE STREQUAL "avif-vendored")
        include("${libavif_BINARY_DIR}/selection.cmake")
        if(NOT FCG_LIBAVIF_BUILT_HERE)
            message(FATAL_ERROR "AVIF source provenance was lost")
        endif()
        if(MODE STREQUAL "avif-system" OR MODE STREQUAL "avif-codec-provided")
            if(NOT selected_aom STREQUAL "SYSTEM")
                message(FATAL_ERROR "Installed/provided AOM was not reused")
            endif()
        elseif(MODE STREQUAL "avif-dav1d" OR MODE STREQUAL "avif-overrides" OR MODE STREQUAL "avif-provided-dav1d"
            OR MODE STREQUAL "avif-alternate-encoder")
            if(NOT selected_dav1d STREQUAL "SYSTEM")
                message(FATAL_ERROR "dav1d selection was lost")
            endif()
        elseif(MODE STREQUAL "avif-gav1")
            if(NOT selected_gav1 STREQUAL "SYSTEM")
                message(FATAL_ERROR "libgav1 selection was lost")
            endif()
        elseif(NOT selected_aom STREQUAL "LOCAL")
            message(FATAL_ERROR "Missing codec did not select local AOM")
        endif()
        if(MODE STREQUAL "avif-alternate-encoder" AND selected_aom)
            message(FATAL_ERROR "Explicit alternate encoder unnecessarily selected AOM")
        endif()
        if(MODE STREQUAL "avif-save" OR MODE STREQUAL "avif-alternate-encoder")
            if(NOT selected_encode)
                message(FATAL_ERROR "Saving did not enable the encoder")
            endif()
        elseif(selected_encode)
            message(FATAL_ERROR "Loading-only fallback enabled the encoder")
        endif()
        if(MODE STREQUAL "avif-shared")
            if(NOT selected_shared)
                message(FATAL_ERROR "Explicit dynamic codec linkage was ignored")
            endif()
        elseif(selected_shared)
            message(FATAL_ERROR "Default libavif linkage was not static")
        endif()
        if(MODE STREQUAL "avif-overrides" AND NOT selected_cpu STREQUAL "fixture-cpu")
            message(FATAL_ERROR "Caller CPU override was shadowed")
        endif()
    elseif(NOT MODE MATCHES "^avif-real" AND FCG_LIBAVIF_BUILT_HERE)
        message(FATAL_ERROR "A supplied/package/vendored codec was source-built")
    endif()
endif()
set(resolved_image_provenance "${FCG_SDL_IMAGE_BUILT_HERE}")
get_target_property(formats fcg-sdl-image FCG_SDL_IMAGE_FORMATS)
if(MODE STREQUAL "provided-empty")
    if(NOT formats STREQUAL "")
        message(FATAL_ERROR "Explicitly empty capability declaration was ignored")
    endif()
elseif(MODE MATCHES "^provided")
    if(NOT formats STREQUAL "BMP;PNG;ANI")
        message(FATAL_ERROR "Supplied target capabilities were lost: ${formats}")
    endif()
elseif(MODE MATCHES "^package")
    if(NOT formats STREQUAL "PNG;WEBP")
        message(FATAL_ERROR "Imported package metadata was lost or confused with requested options: ${formats}")
    endif()
elseif(MODE MATCHES "^avif-")
    if(NOT "AVIF" IN_LIST formats)
        message(FATAL_ERROR "Resolved AVIF was not advertised")
    endif()
elseif(MODE STREQUAL "unresolved")
    if("AVIF" IN_LIST formats)
        message(FATAL_ERROR "Requested but unresolved codec was advertised")
    endif()
elseif(MODE STREQUAL "disabled")
    foreach(codec PNG TGA BMP)
        if(codec IN_LIST formats)
            message(FATAL_ERROR "Disabled ${codec} was advertised")
        endif()
    endforeach()
elseif(MODE MATCHES "^png-")
    if("PNG" IN_LIST formats)
        message(FATAL_ERROR "Disabled PNG was advertised")
    endif()
elseif(MODE STREQUAL "normal" OR MODE STREQUAL "cache")
    if(NOT "WEBP" IN_LIST formats)
        message(FATAL_ERROR "Enabled WebP was not advertised")
    endif()
endif()

# A second include and all subsequent library configuration must consume the resolved dependencies.
macro(CPMAddPackage)
    message(FATAL_ERROR "Library configuration attempted CPMAddPackage after dependency startup")
endmacro()
macro(CPMFindPackage)
    message(FATAL_ERROR "Library configuration attempted CPMFindPackage after dependency startup")
endmacro()
macro(find_package)
    message(FATAL_ERROR "Library configuration attempted find_package after dependency startup")
endmacro()
include("${FCG_SOURCE}/cmake/Dependencies.cmake")
include("${FCG_SOURCE}/cmake/Shaders.cmake")
add_subdirectory("${FCG_SOURCE}/core" core)
add_subdirectory("${FCG_SOURCE}/libs" libs)
if(NOT FCG_SDL_IMAGE_BUILT_HERE STREQUAL resolved_image_provenance OR NOT FCG_IMGUI_BUILT_HERE STREQUAL "OFF")
    message(FATAL_ERROR "Repeated startup or library setup changed dependency provenance")
endif()
if(NOT "Image" IN_LIST FCG_ADDITIONAL_LIBS)
    message(FATAL_ERROR "The intermediate directory did not register Image")
endif()

# Configure-time export validation covers provided targets as well as source-built backends.
install(TARGETS Core Image Extras EXPORT FixtureTargets)
if(NOT BUILD_SHARED_LIBS)
    install(TARGETS fcg-shaders EXPORT FixtureTargets)
endif()
install(EXPORT FixtureTargets NAMESPACE FCG-framework:: DESTINATION lib/cmake/Fixture)
if(MODE MATCHES "^provided|^package")
    return()
endif()
if(MODE MATCHES "^avif-" AND NOT MODE MATCHES "^avif-real")
    return()
endif()
get_target_property(definitions SDL3_image::SDL3_image COMPILE_DEFINITIONS)
if(MODE STREQUAL "minimal")
    foreach(codec AVIF JXL TIF WEBP)
        if("LOAD_${codec}" IN_LIST definitions)
            message(FATAL_ERROR "Minimal defaults enabled external codec ${codec}")
        endif()
    endforeach()
    if("SDL_IMAGE_LIBPNG" IN_LIST definitions)
        message(FATAL_ERROR "Minimal defaults enabled libpng")
    endif()
    add_executable(backend-consumer "${FCG_SOURCE}/libs/image/tests/global.cpp")
    target_link_libraries(backend-consumer PRIVATE FCG-framework::Image)
    set(test_consumer backend-consumer)
elseif(MODE MATCHES "^avif-real")
    add_executable(optional-codec "${CMAKE_CURRENT_LIST_DIR}/optional.cpp")
    target_link_libraries(optional-codec PRIVATE FCG-framework::Image)
    set(test_consumer optional-codec)
elseif(MODE MATCHES "^png-" OR MODE STREQUAL "disabled" OR MODE STREQUAL "unresolved")
    if(MODE STREQUAL "unresolved")
        return()
    endif()
    if("LOAD_PNG" IN_LIST definitions OR SDLIMAGE_PNG)
        message(FATAL_ERROR "Caller's built-in codec override was shadowed")
    endif()
else()
    if(NOT "LOAD_WEBP" IN_LIST definitions)
        message(FATAL_ERROR "Caller's external codec override was shadowed")
    endif()
    add_executable(optional-codec "${CMAKE_CURRENT_LIST_DIR}/optional.cpp")
    target_link_libraries(optional-codec PRIVATE FCG-framework::Image)
    set(test_consumer optional-codec)
endif()
if(test_consumer)
    if(TEST_BACKEND_SHARED)
        add_custom_command(TARGET ${test_consumer} POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:SDL3::SDL3>"
                "$<TARGET_FILE_DIR:${test_consumer}>")
    endif()
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/consumer-$<CONFIG>.cmake"
        CONTENT "set(consumer \"$<TARGET_FILE:${test_consumer}>\")\n")
endif()
