cmake_minimum_required(VERSION 3.31)
project(OpenEXR VERSION 3.4.14 LANGUAGES CXX)
add_library(OpenEXR::OpenEXR INTERFACE IMPORTED GLOBAL)
if(NOT TARGET Imath::Imath OR BUILD_SHARED_LIBS OR NOT CMAKE_POSITION_INDEPENDENT_CODE)
    message(FATAL_ERROR "EXR fallback requires resolved Imath and static PIC configuration")
endif()
if(OPENEXR_BUILD_EXAMPLES OR OPENEXR_BUILD_PYTHON OR OPENEXR_TEST_LIBRARIES OR OPENEXR_INSTALL)
    message(FATAL_ERROR "Unrequested EXR extras enabled")
endif()
if(MODE STREQUAL "exr-overrides")
    if(NOT OPENEXR_BUILD_TOOLS)
        message(FATAL_ERROR "Caller EXR option was shadowed")
    endif()
elseif(OPENEXR_BUILD_TOOLS)
    message(FATAL_ERROR "EXR tools enabled by default")
endif()
