cmake_minimum_required(VERSION 3.31)
project(ImageDependencyFixture LANGUAGES C CXX)
include("${FCG_SOURCE}/cmake/CPM.cmake")
include(GenerateExportHeader)
set(FCG_PROJECT_NAME FCG-framework)
set(FCG_FRAMEWORK_VERSION 0.0.1)
set(FCG_FRAMEWORK_SOVERSION 0)
set(FCG_CXX_STANDARD cxx_std_23)
set(FCG_USE_SHARED_SDL OFF)
set(BUILD_SHARED_LIBS OFF)
set(CPM_USE_LOCAL_PACKAGES OFF)
set(CPM_SDL3_image_SOURCE "${IMAGE_DEPENDENCY_SOURCE}")
find_package(SDL3 3.4 CONFIG REQUIRED)

if(MODE STREQUAL "provided")
    # A provided target must short-circuit lookup/fetching and leave upstream options alone.
    add_library(provided INTERFACE)
    add_library(SDL3_image::SDL3_image ALIAS provided)
    set(SDLIMAGE_WEBP ON)
    include("${FCG_SOURCE}/image/Dependencies.cmake")
    if(NOT SDLIMAGE_WEBP OR FCG_SDL_IMAGE_BUILT_HERE)
        message(FATAL_ERROR "Provided dependency was reconfigured")
    endif()
    return()
elseif(MODE STREQUAL "normal")
    set(SDLIMAGE_WEBP ON)
    set(SDLIMAGE_STRICT ON)
elseif(MODE STREQUAL "cache")
    set(SDLIMAGE_WEBP ON CACHE BOOL "Exercise a parent's cached codec option")
    set(SDLIMAGE_STRICT ON CACHE BOOL "Require the opted-in codec")
endif()

add_subdirectory("${FCG_SOURCE}/image" image)
if(MODE STREQUAL "minimal")
    get_target_property(definitions SDL3_image::SDL3_image COMPILE_DEFINITIONS)
    foreach(codec AVIF JXL TIF WEBP)
        if("LOAD_${codec}" IN_LIST definitions)
            message(FATAL_ERROR "Minimal defaults enabled external codec ${codec}")
        endif()
    endforeach()
    if("SDL_IMAGE_LIBPNG" IN_LIST definitions)
        message(FATAL_ERROR "Minimal defaults enabled libpng")
    endif()
elseif(MODE STREQUAL "normal" OR MODE STREQUAL "cache")
    get_target_property(definitions SDL3_image::SDL3_image COMPILE_DEFINITIONS)
    if(NOT "LOAD_WEBP" IN_LIST definitions)
        message(FATAL_ERROR "Caller codec option was shadowed")
    endif()
    add_executable(optional-codec "${CMAKE_CURRENT_LIST_DIR}/optional.cpp")
    target_link_libraries(optional-codec PRIVATE FCG-framework::Image)
else()
    message(FATAL_ERROR "Unknown fixture mode")
endif()
